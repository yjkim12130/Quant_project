  221 def add_ind(data):
  222     df=data.copy();p=df["Price"];df["MA_200"]=p.rolling(200).mean()
  223     df["Disp"]=(p-df["MA_200"])/df["MA_200"]*100
  224     sma=p.rolling(200).mean();mad=p.rolling(200).apply(lambda x:np.abs(x-x.mean()).mean(),raw=True)
  225     df["CCI"]=(p-sma)/(0.015*mad)
  226     d=p.diff();g=d.where(d>0,0.0).rolling(14).mean();l=(-d.where(d<0,0.0)).rolling(14).mean()
  227     df["RSI"]=100-(100/(1+g/l));df["MDD"]=(p/p.rolling(252,min_periods=1).max())-1.0
  228     return df

  243 # ★ [#13] staged 파라미터 추가 — Panic 단계매집 지원
  244 def run_beta(data, params, budget, inh=None, sim_start=None, sim_end=None, staged=False):
  245     """
  246     staged=False (기본): v8.7 반복매집 — panic_period마다 조건 충족 시 반복 매수
  247     staged=True  (신규): v8.8 단계매집 — MDD 구간별 최초 1회만 매수 (1차→2차→3차)
  248     """
  249     df=add_ind(data);df=df.dropna(subset=["MA_200","MDD"]).copy()
  250     if sim_start:df=df[df.index>=pd.Timestamp(sim_start)]
  251     if sim_end:df=df[df.index<=pd.Timestamp(sim_end)]
  252     if df.empty:return _emp()
  253     ms=list(params["mdd_threshold_sets"]);ds=list(params["drain_sets"])
  254     dst=(budget*(params["stock_ratio"]/18))/TRADING_DAYS;dcs=(budget*((18-params["stock_ratio"])/18))/TRADING_D
      AYS
  255     if inh:
  256         ys=float(inh.get("yj_st",0));yc=float(inh.get("yj_cs",0));cur=inh.get("state","Bull")
  257         pc=int(inh.get("p_cnt",0));tp=float(inh.get("total_principal",0))
  258         panic_stage=int(inh.get("panic_stage",0))  # ★ 단계매집용 이전 단계 기억
  259     else:
  260         ys=yc=0.0;cur="Bull";pc=0;tp=0.0
  261         panic_stage=0  # ★ 0=미매수, 1=1차완료, 2=2차완료, 3=3차완료
  262     recs,slogs,plogs,elogs=[],[],[],[];pt=0.0;panic_ep=None
  263     for dt,row in df.iterrows():
  264         dr=float(row.get("Daily_Ret",0) or 0);vx=float(row.get("VIX",15));cc=float(row.get("CCI",0))
  265         rs=float(row.get("RSI",50));md=float(row.get("MDD",0));dp=float(row.get("Disp",0))
  266         pv=cur;cur=_state(cur,vx,cc,rs,md,dp,params)
  267         lv={"Bull":params["lev_bull"],"Bear":params["lev_bear"],"Panic":params["lev_panic"]}[cur]
  268         if cur!=pv:slogs.append({"날짜":dt.date(),"변화":f"{pv}→{cur}","VIX":round(vx,1),"MDD%":round(md*100,1)
      ,"CCI":round(cc,1),"RSI":round(rs,1),"Disp%":round(dp,1),"레버리지":lv})
  269         # ★ Panic 탈출 시 단계 초기화
  270         if pv=="Panic" and cur!="Panic":
  271             panic_stage=0
  272         tp+=budget/TRADING_DAYS
  273         if dr:ys*=1+dr*lv-max(0,lv-1)*(LEV_COST/TRADING_DAYS)
  274         yc*=1+CMA_INTEREST/TRADING_DAYS;ys+=dst;yc+=dcs
  275         if cur=="Panic":
  276             if panic_ep is None:panic_ep={"시작":dt.date(),"종료":"-","지속일":0,"전CMA(만)":round(yc/1e4,1),"
