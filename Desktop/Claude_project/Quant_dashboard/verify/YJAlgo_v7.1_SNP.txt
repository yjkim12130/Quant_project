# ==========================================
start_year = 2010
end_year = 2026
annual_budget = 40000000  # 연간 투자금 4,000만 원
daily_total_budget = annual_budget / 252
lev_cost_annual = 0.045   # 레버리지 비용 (4.5%)
cma_interest_annual = 0.035 # CMA 이자 (3.5%)

# S&P 500에 맞춘 파라미터 그리드
param_grid = {
    'lev_bull': [2.0], 'lev_bear': [1.0], 'lev_panic': [1.5],
    'vix_panic': [25],
    'mdd_panic': [-0.12, -0.15],
    'cci_bull': [50, 80, 100], 'cci_bear': [-50, -100],
    'rsi_bull': [50, 70, 90], 'rsi_bear': [40, 60, 80],
    'stock_ratio': [15], 'panic_period': [1, 5], 'periodic_denom': [1, 5],
    'mdd_threshold_sets': [(0.45, 0.30, 0.15)],
    'drain_sets': [(1.0, 0.6, 0.4)]
}

# ==========================================
# 1. 데이터 로딩 및 지표 계산
# ==========================================
print(f"⏳ 시장 데이터를 불러오고 지표를 계산 중입니다... ({start_year}~2026)")

    'Nasdaq': raw_df[('Close', '^IXIC')],   # 보조: 나스닥
    'VIX': raw_df[('Close', '^VIX')]        # 공포: S&P 500 VIX
})
df = df.ffill()
df['VIX'] = df['VIX'].fillna(15)

def add_indicators(data, price_col, prefix=''):
    data[f'{prefix}MA_200'] = data[price_col].rolling(200).mean()
    data[f'{prefix}Disp'] = (data[price_col] - data[f'{prefix}MA_200']) / data[f'{prefix}MA_200'] * 100
    tp = data[price_col]
    data[f'{prefix}CCI'] = (tp - tp.rolling(200).mean()) / (0.015 * tp.rolling(200).apply(lambda x: np.abs(x - x.mean()).mean()))
    delta = data[price_col].diff()
    gain = (delta.where(delta > 0, 0)).rolling(14).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(14).mean()
    data[f'{prefix}RSI'] = 100 - (100 / (1 + (gain / loss)))
    data[f'{prefix}MDD'] = (data[price_col] / data[price_col].rolling(252, min_periods=1).max()) - 1.0
    return data

df = add_indicators(df, 'Price', '')
df = add_indicators(df, 'Nasdaq', 'NDQ_')
df['Daily_Ret'] = df['Price'].pct_change()
df = df.dropna(subset=['MA_200', 'MDD']).copy()
df = df[df.index.year >= start_year]

# ==========================================
# 2. 시뮬레이션 엔진 함수 (상태 변경 로그 추가)
# ==========================================
def run_simulation(params, data_df):
    yj_st, yj_cs = 0.0, 0.0
    dca_p, dca_sh = 0.0, 0.0
    cur_s = 'Bull'
    history, panic_logs, s_logs = [], [], []
    panic_ep = None
    p_buy_cnt = 0

    d_st_base = (annual_budget * (params['stock_ratio'] / 18)) / 252
    d_cs_base = (annual_budget * ((18 - params['stock_ratio']) / 18)) / 252
    m_sets, d_sets = params['mdd_threshold_sets'], params['drain_sets']

    for date, row in data_df.iterrows():
        new_s = cur_s
        if row['VIX'] >= params['vix_panic'] and row['MDD'] <= params['mdd_panic']:
            new_s = 'Panic'
        else:
            if cur_s == 'Panic':
                if row['CCI'] > 0 and row['RSI'] > params['rsi_bear']: new_s = 'Bear'
            elif cur_s == 'Bear':
                if row['CCI'] > params['cci_bull'] and row['RSI'] > params['rsi_bull']: new_s = 'Bull'
            else:
                if row['CCI'] < params['cci_bear'] and row['Disp'] < -3.0: new_s = 'Bear'

        # 상태가 변경될 경우 로그 기록
        if new_s != cur_s:
            s_logs.append({
                'Date': date, 'From': cur_s, 'To': new_s,
                'VIX': row['VIX'], 'MDD': row['MDD'], 'CCI': row['CCI'], 'RSI': row['RSI']
            })
        cur_s = new_s

        lev = params['lev_bull'] if cur_s == 'Bull' else (params['lev_panic'] if cur_s == 'Panic' else params['lev_bear'])

        if yj_st > 0:
            yj_st *= (1 + (row['Daily_Ret'] * lev) - max(0, lev - 1) * (lev_cost_annual / 252))
        yj_cs *= (1 + cma_interest_annual / 252)

        dca_p += daily_total_budget
        dca_sh += daily_total_budget / row['Price']
        yj_st += d_st_base
        yj_cs += d_cs_base

        p_buy = 0
        if cur_s == 'Panic':
            if panic_ep is None:
                panic_ep = {'Start': date, 'Total_Buy': 0.0, 'Max_MDD': row['MDD'], 'Pre_CMA': yj_cs, 'Days': 0}
            if p_buy_cnt % params['panic_period'] == 0:
                m = abs(row['MDD'])
                dr = d_sets[0] if m >= m_sets[0] else (d_sets[1] if m >= m_sets[1] else (d_sets[2] if m >= m_sets[2] else 0))
                p_buy = min((yj_cs * dr) / params['periodic_denom'], yj_cs)
                yj_st += p_buy; yj_cs -= p_buy; panic_ep['Total_Buy'] += p_buy
            panic_ep['Max_MDD'] = min(panic_ep['Max_MDD'], row['MDD'])
            panic_ep['Days'] += 1; p_buy_cnt += 1
        else:
            if panic_ep is not None:
                panic_ep['End'] = date; panic_ep['Post_CMA'] = yj_cs; panic_logs.append(panic_ep); panic_ep = None
            p_buy_cnt = 0

        history.append([date, dca_p, dca_sh * row['Price'], yj_st + yj_cs, yj_st, yj_cs, cur_s, lev, p_buy])

    res_df = pd.DataFrame(history, columns=['Date', 'DCA_Prin', 'DCA_Val', 'YJ_Val', 'YJ_Stock', 'YJ_Cash', 'State', 'Lev', 'Panic_Buy']).set_index('Date')
    return res_df, panic_logs, s_logs
